cmake_minimum_required(VERSION 3.20)
project(REWorkbench VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_subdirectory(native/bytecore_cpp)
add_subdirectory(native/workbench)
option(REWORKBENCH_BUILD_TEST_FIXTURES "Build self-authored native debugger test targets" OFF)
if(REWORKBENCH_BUILD_TEST_FIXTURES)
    add_executable(reworkbench-test-target tests/native/debug_target.cpp)
endif()
